/*CREACIÓN DE LA BD*/
DROP DATABASE IF EXISTS bd_tema7;
CREATE DATABASE bd_tema7 CHARACTER SET utf8mb4 COLLATE 'utf8mb4_spanish_ci';
USE bd_tema7;

/*CREACIÓN DE LAS TABLAS*/
Create TABLE ALUM (NOMBRE VARCHAR(20), EDAD TINYINT, LOCALIDAD VARCHAR(15));
Create TABLE NUEVOS (NOMBRE VARCHAR(20), EDAD TINYINT, LOCALIDAD VARCHAR(15));
Create TABLE ANTIGUOS (NOMBRE VARCHAR(20), EDAD TINYINT, LOCALIDAD VARCHAR(15));
CREATE TABLE PERSONAL (COD_CENTRO DECIMAL(4,0) NOT NULL, DNI DECIMAL(10,0), APELLIDOS VARCHAR(30), FUNCION VARCHAR(15), SALARIO DECIMAL(10,0));
CREATE TABLE PROFESORES (COD_CENTRO DECIMAL(4,0) NOT NULL, DNI DECIMAL(10,0), APELLIDOS VARCHAR(30), ESPECIALIDAD VARCHAR(16));
CREATE TABLE CENTROS (COD_CENTRO DECIMAL(4,0) NOT NULL, TIPO_CENTRO CHAR(1), NOMBRE VARCHAR(30), DIRECCION VARCHAR(26), TELEFONO VARCHAR(10), NUM_PLAZAS DECIMAL(4,0));
Create TABLE BANCOS(COD_BANCO DECIMAL(4,0) PRIMARY KEY, NF_BANCO VARCHAR(10), NOMBRE_BANC VARCHAR(30), DOM_FISCAL VARCHAR(35), POBLACION VARCHAR(35));
Create TABLE SUCURSALES(COD_BANCO DECIMAL(4,0), COD_SUCUR DECIMAL(4,0), NOMBRE_SUC VARCHAR(35), DIREC_SUC VARCHAR(35), LOC_SUC VARCHAR(20), PROV_SUC VARCHAR(20), PRIMARY KEY (COD_BANCO,COD_SUCUR));	
Create TABLE CUENTAS(COD_BANCO DECIMAL(4,0), COD_SUCUR DECIMAL(4,0), NUM_CTA DECIMAL(10,0), FECHA_ALTA DATE, NOMBRE_CTA VARCHAR(35), DIREC_CTA VARCHAR(35), POBLA_CTA VARCHAR(20), SALDO_DEBE DECIMAL(10,0), SALDO_HABER DECIMAL(10,0), PRIMARY KEY (COD_BANCO,COD_SUCUR,NUM_CTA));
CREATE TABLE MOVIMIENTOS(COD_BANCO DECIMAL(4,0), COD_SUCUR DECIMAL(4,0), NUM_CTA DECIMAL(10,0), FECHA_MOV DATE, TIPO_MOV CHAR(1), IMPORTE DECIMAL(10,0));
