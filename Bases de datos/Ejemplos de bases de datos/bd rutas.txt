DROP DATABASE IF EXISTS bd_rutas;

CREATE OR REPLACE DATABASE bd_rutas CHARACTER SET UTF8MB4 COLLATE 'utf8mb4_spanish_ci';

USE bd_rutas;

CREATE OR REPLACE TABLE persona(
	cod INT,
	nombre VARCHAR(20),
	apellido1 VARCHAR(20),
	apellido2 VARCHAR(20),
	email VARCHAR(30),
	CONSTRAINT per_cod_pk PRIMARY KEY(cod)
);

CREATE OR REPLACE TABLE comentario(
	cod_comentario INT,
	cod_persona INT,
	CONSTRAINT com_cod_pk PRIMARY KEY(cod_comentario,cod_persona),
	CONSTRAINT com_cod_fk FOREIGN KEY(cod_persona) REFERENCES persona(cod) ON DELETE CASCADE 
);

CREATE OR REPLACE TABLE localidad(
	nombre VARCHAR(20),
	rio VARCHAR(20),
	CONSTRAINT loc_nri_pk PRIMARY KEY(nombre,rio)
);

CREATE OR REPLACE TABLE ruta(
	nombre VARCHAR(50),
	descripcion VARCHAR(70),
	distancia DOUBLE,
	circular BOOLEAN, 
	tipo_ruta ENUM('P', 'B', 'C'),
	nom_loc_ini VARCHAR(20),
	rio_loc_ini VARCHAR(20),
	nom_loc_fin VARCHAR(20),
	rio_loc_fin VARCHAR(20),
	CONSTRAINT rut_nom_pk PRIMARY KEY(nombre),
	CONSTRAINT rut_nr1_fk FOREIGN KEY(nom_loc_ini,rio_loc_ini) REFERENCES localidad(nombre,rio) ON DELETE CASCADE,
	CONSTRAINT rut_nr2_fk FOREIGN KEY(nom_loc_fin,rio_loc_fin) REFERENCES localidad(nombre,rio) ON DELETE CASCADE
);


CREATE OR REPLACE TABLE ruta_localidad(
	nombre_localidad VARCHAR(20),
	nombre_ruta VARCHAR(50),
	CONSTRAINT rlo_nom_pk PRIMARY KEY(nombre_localidad,nombre_ruta),
	CONSTRAINT rlo_loc_fk FOREIGN KEY(nombre_localidad) REFERENCES localidad(nombre) ON DELETE CASCADE,
	CONSTRAINT rlo_rut_fk FOREIGN KEY(nombre_ruta) REFERENCES ruta(nombre) ON DELETE CASCADE
);

CREATE OR REPLACE TABLE ruta_persona(
	nombre_ruta VARCHAR(50),
	cod_persona INT,
	CONSTRAINT rpe_nco_pk PRIMARY KEY(nombre_ruta,cod_persona),
	CONSTRAINT rpe_nom_fk FOREIGN KEY(nombre_ruta) REFERENCES ruta(nombre) ON DELETE CASCADE,
	CONSTRAINT rpe_cod_fk FOREIGN KEY(cod_persona) REFERENCES persona(cod) ON DELETE CASCADE
);

CREATE OR REPLACE TABLE a_pie(
	nombre_ruta VARCHAR(50),
	altitud DOUBLE,
	dificultad ENUM('Básico','Medio','Alto','Muy alto'),
	CONSTRAINT api_nom_fk FOREIGN KEY(nombre_ruta) REFERENCES ruta(nombre) ON DELETE CASCADE,
	CONSTRAINT api_nom_pk PRIMARY KEY(nombre_ruta)
);

CREATE OR REPLACE TABLE bicicleta(
	nombre_ruta VARCHAR(20),
	altitud DOUBLE,
	dificultad ENUM('Básico','Medio','Alto','Muy alto'),
	CONSTRAINT bic_nom_fk FOREIGN KEY(nombre_ruta) REFERENCES ruta(nombre) ON DELETE CASCADE,
	CONSTRAINT bic_nom_pk PRIMARY KEY(nombre_ruta)
);

CREATE OR REPLACE TABLE coche(
	nombre_ruta VARCHAR(50),
	costo_gasolina DOUBLE,
	litros_consumidos DOUBLE,
	emision_gasolina DOUBLE,
	CONSTRAINT coc_nom_fk FOREIGN KEY(nombre_ruta) REFERENCES ruta(nombre) ON DELETE CASCADE,
	CONSTRAINT coc_nom_pk PRIMARY KEY(nombre_ruta)
);
