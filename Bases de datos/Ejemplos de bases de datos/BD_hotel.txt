/*----------Elimina la base de datos anterior----------*/
DROP DATABASE IF EXISTS bd_hotel;

/*---------Crea la base de datos----------*/
CREATE DATABASE IF NOT EXISTS bd_hotel CHARACTER SET UTF8MB4 COLLATE 'utf8mb4_spanish_ci';

/*---------Usa la base de datos----------*/
USE bd_hotel;

/*----------Crea las tablas de la base de datos----------*/
CREATE OR REPLACE TABLE Hotel(
    ID_Hotel INT,
    Nombre VARCHAR(20),
    Teléfono VARCHAR(20),
    Ubicación VARCHAR(30)
	);

CREATE OR REPLACE TABLE Empleado(
    ID_Empleado INT,
    Nombre VARCHAR(20),
    Apellido VARCHAR(20),
    Teléfono VARCHAR(20),
    Salario INT,
    Tipo VARCHAR(20),
    ID_Hotel INT
	);

CREATE OR REPLACE TABLE Habitacion(
    ID_habitación INT,
    Precio INT,
    Descripcion VARCHAR(100), 
    Tipo VARCHAR(10),
    ID_Hotel INT
	);

CREATE OR REPLACE TABLE Cliente(
    ID_Cliente INT,
    Nombre VARCHAR(20),
    Apellido VARCHAR(20),
    Teléfono VARCHAR(20)
	);

CREATE OR REPLACE TABLE Reserva(
    ID_Reserva INT,
    Fecha_Entrada DATE,
    Fecha_Salida DATE,
    ID_Cliente INT,
    ID_Habitación INT
	);

CREATE OR REPLACE TABLE Realiza_Reserva(
    ID_Recepcionista INT,
    ID_Cliente INT,
    ID_Reserva INT,
    Fecha_Hora DATE
	);

/*----------Restricciones a Hotel----------*/
ALTER TABLE hotel
ADD CONSTRAINT hot_id_pk PRIMARY KEY (ID_Hotel);

/*---------Restricciones a Empleado----------*/
ALTER TABLE empleado
ADD CONSTRAINT emp_id_pk PRIMARY KEY (ID_Empleado),
ADD CONSTRAINT emp_hot_fk FOREIGN KEY (ID_Hotel)
REFERENCES hotel(ID_Hotel) ON DELETE SET NULL ON UPDATE CASCADE; 

/*--------Restricciones a Habitación----------*/
ALTER TABLE habitacion 
ADD CONSTRAINT hab_id_pk PRIMARY KEY (ID_habitación),
ADD CONSTRAINT hab_hot_fk FOREIGN KEY (ID_Hotel) 
REFERENCES hotel(ID_Hotel) ON DELETE SET NULL ON UPDATE CASCADE;

/*--------Restricciones a Cliente------------*/
ALTER TABLE cliente
ADD CONSTRAINT cli_id_pk PRIMARY KEY (ID_Cliente);

/*--------Restricciones a Reserva------------*/
ALTER TABLE reserva
ADD CONSTRAINT res_id_pk PRIMARY KEY (ID_Reserva),
ADD CONSTRAINT res_cli_fk FOREIGN KEY (ID_Cliente)
REFERENCES cliente(ID_Cliente) ON DELETE SET NULL ON UPDATE CASCADE,
ADD CONSTRAINT res_hab_fk FOREIGN KEY (ID_habitación)
REFERENCES habitacion(ID_habitación) ON DELETE SET NULL ON UPDATE CASCADE,
ADD CONSTRAINT res_fec_ck CHECK (Fecha_Entrada<Fecha_Salida);

/*-------Restricciones a Realiza_Reserva--------*/
ALTER TABLE realiza_reserva
ADD CONSTRAINT rea_rec_fk FOREIGN KEY (ID_Recepcionista)
REFERENCES empleado(ID_Empleado) ON DELETE SET NULL ON UPDATE CASCADE,
ADD CONSTRAINT rea_cli_fk FOREIGN KEY (ID_Cliente)
REFERENCES cliente(ID_Cliente) ON DELETE SET NULL ON UPDATE CASCADE,
ADD CONSTRAINT rea_hab_fk FOREIGN KEY (ID_Reserva)
REFERENCES reserva(ID_Reserva) ON DELETE SET NULL ON UPDATE CASCADE;

/*Clientes atendidos por un recepcionista en específico*/
CREATE OR REPLACE VIEW Consulta1 AS 
SELECT c.Nombre, c.Apellido
FROM cliente c INNER JOIN realiza_reserva r ON c.ID_Cliente=r.ID_Cliente
INNER JOIN empleado e ON e.ID_Empleado=r.ID_Recepcionista
WHERE r.ID_Recepcionista IN (SELECT e2.ID_Empleado
									  FROM empleado e2
									  WHERE LCASE(e2.Tipo)="recepcion"
									  AND LCASE(e2.Nombre)="mateo"
									  AND LCASE(e2.Apellido)="sanchez")
;

/*Consuta con Group By*/
CREATE OR REPLACE VIEW Group_by AS 
SELECT DISTINCT(h.Tipo)
FROM habitacion h 
WHERE h.Tipo IN (SELECT h2.Tipo
				  	  FROM habitacion h2 INNER JOIN reserva r2 ON h2.ID_habitación=r2.ID_Habitación
				     GROUP BY h2.Tipo
				     HAVING COUNT(*)=(SELECT COUNT(*)
				  						   FROM habitacion h3 INNER JOIN reserva r3 ON h3.ID_habitación=r3.ID_Habitación
				  						   GROUP BY h3.Tipo
											ORDER BY COUNT(*)DESC
											LIMIT 1))
;

/*Se necesita saber las habitaciones disponibles en una fecha y hotel en especifico*/

SELECT h.*
FROM habitacion h
WHERE h.ID_Hotel=101 AND h.ID_habitación NOT IN (SELECT r.ID_Habitación
																 FROM Reserva r
																 WHERE r.Fecha_Entrada>='2024-05-20' AND r.Fecha_Salida<='2024-05-27')
;
