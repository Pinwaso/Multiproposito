CREATE DATABASE Hospital;
USE Hospital;
CREATE TABLE HOSPITAL
(
HOSPITAL_COD INT (2) NOT NULL PRIMARY KEY,
NOMBRE VARCHAR (10),
DIRECCION VARCHAR (20),
TELEFONO VARCHAR (8),
NUM_CAMA INT(3)
);
CREATE TABLE SALA
(
HOSPITAL_COD INT(2) NOT NULL,
SALA_COD INT (2) NOT NULL,
NOMBRE VARCHAR (20),
NUM_CAMA INT(3)
);
CREATE TABLE PLANTILLA
(
HOSPITAL_COD INT(2) NOT NULL,
SALA_COD INT (2) NOT NULL,
EMPLEADO_NO INT (4) NOT NULL,
APELLIDO VARCHAR (15),
FUNCION VARCHAR (10),
TURNO VARCHAR (1),
SALARIO INT (10)
);
CREATE TABLE OCUPACION
(
INSCRIPCION INT (5) NOT NULL,
HOSPITAL_COD INT(2) NOT NULL,
SALA_COD INT (2) NOT NULL,
CAMA INT (4)
);
CREATE TABLE DOCTOR
(
HOSPITAL_COD INT(2) NOT NULL,
DOCTOR_NO INT (3) NOT NULL,
APELLIDO VARCHAR (13),
ESPECIALIDAD VARCHAR (16)
);
CREATE TABLE ENFERMO
(
INSCRIPCION INT (5) NOT NULL PRIMARY KEY,
APELLIDO VARCHAR (15),
DIRECCION VARCHAR (20),
FECHA_NAC DATE,
S VARCHAR (1),
NSS INT (9)
);
CREATE TABLE EMP
(
EMP_NO INT (2) NOT NULL,
APELLIDO VARCHAR (10),
OFICIO VARCHAR (10),
DIR INT (4),
FECHA_ALTA DATE,
SALARIO INT (10),
COMISION INT (10),
DEPT_NO INT (2) NOT NULL
);
CREATE TABLE DEPT2
(
DEPT_NO INT (2) NOT NULL PRIMARY KEY,
DNOMBRE VARCHAR (14),
LOC VARCHAR (14)
);
