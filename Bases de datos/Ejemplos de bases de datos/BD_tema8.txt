DROP DATABASE IF EXISTS BIBLIO;
CREATE DATABASE BIBLIO;
USE BIBLIO;

-- ------------------------------------------------------
--  DDL for Table LECTOR
-- ------------------------------------------------------
  CREATE TABLE LECTOR 
   (NUMLECTOR DECIMAL(6,0) PRIMARY KEY, 
	NOMLECTOR VARCHAR(30), 
	TIPOLECTOR DECIMAL(6,0)
   );

-- ------------------------------------------------------
--  DDL for Table LIBRO
-- ------------------------------------------------------
CREATE TABLE LIBRO 
   (	 CODLIBRO  DECIMAL(6,0) PRIMARY KEY, 
	 TITULO  VARCHAR(30), 
	 NUMPAG  DECIMAL(4,0), 
	 EDITORIAL  VARCHAR(20), 
	 AUTOR  VARCHAR(30)
   ) ;

-- ------------------------------------------------------
--  DDL for Table PRESTAMO
-- ------------------------------------------------------

CREATE TABLE  PRESTAMO  
   (	 CODLIBRO  DECIMAL(6,0) , 
	 NUMLECTOR  DECIMAL(6,0) , 
	 FPRESTA  DATE,
	 DIAS  DECIMAL(3,0),
	CONSTRAINT prestamo_pk PRIMARY KEY (CODLIBRO, NUMLECTOR, FPRESTA),
	 CONSTRAINT codlibrop_fk FOREIGN KEY (CODLIBRO) REFERENCES  LIBRO(CODLIBRO),
	 CONSTRAINT numlectorp_fk FOREIGN KEY (NUMLECTOR) REFERENCES LECTOR(NUMLECTOR)
   ) ;

-- ------------------------------------------------------
--  DDL for Table DEVOLUCION
-- ------------------------------------------------------

CREATE TABLE  DEVOLUCION  
   (CODLIBRO  DECIMAL(6,0) NOT NULL,  
	 NUMLECTOR  DECIMAL(6,0) NOT NULL,  
	 FPRESTA  DATE NOT NULL, 
	 FDEVOL  DATE NOT NULL,
	CONSTRAINT devol_pk PRIMARY KEY (CODLIBRO, NUMLECTOR, FPRESTA, FDEVOL),
	CONSTRAINT codlibrod_fk FOREIGN KEY (CODLIBRO) REFERENCES LIBRO(CODLIBRO),
	CONSTRAINT numlectord_fk FOREIGN KEY (NUMLECTOR) REFERENCES LECTOR(NUMLECTOR)
   ) ;
