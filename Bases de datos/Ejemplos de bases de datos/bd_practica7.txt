DROP DATABASE IF EXISTS db_practica7;
CREATE DATABASE IF NOT EXISTS db_practica7
CHARACTER SET 'utf8mb4';

USE db_practica7;

DROP TABLE if EXISTS ejemplo2;
CREATE TABLE if NOT EXISTS ejemplo2(
	DNI INT UNSIGNED NOT NULL,
	NOMBRE VARCHAR(20) DEFAULT "no definido",
	NUM_USUARIO INT
	);
	
	
DROP TABLE if EXISTS ALUMNOS07;
CREATE TABLE if NOT EXISTS ALUMNOS07(
	NUMERO_MATRICULA INT(6) NOT NULL ,
	NOMBRE VARCHAR(15) NOT NULL,
	FECHA_NACIMIENTO DATE,
	DIRECCION VARCHAR(30),
	LOCALIDAD VARCHAR(15)
	);

DROP TABLE if EXISTS ALUMCOPIA;
CREATE TABLE if not EXISTS ALUMCOPIA(
	DNI VARCHAR(10) NOT NULL,
	NOMBRE VARCHAR(15) NOT NULL,
	APELLIDOS VARCHAR(20) NOT NULL,
	FECHA_NAC DATETIME,
	DIRECCION VARCHAR(20),
	POBLACION VARCHAR(20),
	PROVINCIA VARCHAR(20),
	CURSO TINYINT(4) NOT NULL,
	NIVEL VARCHAR(10) NOT NULL,
	CLASE CHAR(1) NOT NULL,
	FALTAS1 TINYINT(4),
	FALTAS2 TINYINT(4),
	FALTAS3 TINYINT(4)
	);
	

DROP TABLE if EXISTS ALUMÑOS;	
CREATE TABLE if not EXISTS ALUMÑOS(
	DNI VARCHAR(10) NOT NULL,
	NOMBRE VARCHAR(15) NOT NULL,
	APELLIDOS VARCHAR(20) NOT NULL,
	FECHA_NAC DATETIME,
	DIRECCION VARCHAR(20),
	POBLACION VARCHAR(20),
	PROVINCIA VARCHAR(20),
	CURSO TINYINT(4) NOT NULL,
	NIVEL VARCHAR(10) NOT NULL,
	CLASE CHAR(1) NOT NULL,
	FALTAS1 TINYINT(4),
	FALTAS2 TINYINT(4),

DROP TABLE if EXISTS EJEMPLO03;
CREATE TABLE if NOT EXISTS EJEMPLO03(
	DNI VARCHAR(10) PRIMARY KEY NOT NULL,
	NOMBRE VARCHAR(30) NOT NULL,
	EDAD INT(4),
	CURSO INT(4),
	CONSTRAINT ejem03_edad_c CHECK (EDAD BETWEEN 5 AND 20),
	CONSTRAINT ejem03_nombre_c CHECK (NOMBRE=UCASE(NOMBRE)),
	CONSTRAINT ejem03_curso_c CHECK (CURSO BETWEEN 1 AND 3)
	);


DROP TABLE if EXISTS fabricantes2;
CREATE TABLE if NOT EXISTS fabricantes2(
	Cod_fabricante INT(4),
	nombre VARCHAR(15),
	pais VARCHAR(15),
	PRIMARY KEY (Cod_fabricante),
	CONSTRAINT fabri2_nombre_c CHECK (nombre=UCASE(nombre)),
	CONSTRAINT fabri2_pais_c CHECK (pais=UCASE(pais))
	);


DROP TABLE if EXISTS articulos2;
CREATE TABLE if not EXISTS articulos2(
	articulo VARCHAR(20),
	cod_fabricante INT,
	peso DECIMAL(3, 0),
	categoria VARCHAR(10),
	precio_venta DECIMAL(8, 2),
	precio_costo DECIMAL(8, 2),
	existencias INT(4),
	PRIMARY KEY (articulo, cod_fabricante, peso, categoria),
	CONSTRAINT art2_fabricante_fk FOREIGN KEY (cod_fabricante) 
	REFERENCES fabricantes2(Cod_fabricante) ON DELETE CASCADE ON UPDATE CASCADE, 
	CONSTRAINT art2_venta_c CHECK (precio_venta>=0),
	CONSTRAINT art2_costo_c CHECK (precio_costo>=0),
	CONSTRAINT art2_peso_c CHECK (peso>=0),  
	CONSTRAINT art2_categoria_c CHECK (categoria IN ("Primera", "Segunda", "Tercera"))
	);


DROP TABLE if EXISTS pedidos2;
CREATE TABLE if NOT EXISTS pedidos2(
	NIF VARCHAR(10),
	ARTICULO VARCHAR(20),
	COD_FABRICANTE INT,
	PESO DECIMAL(3, 0),
	CATEGORIA VARCHAR(10),
	FECHA_PEDIDO DATE,
	UNIDADES_PEDIDAS INT,
	PRIMARY KEY (NIF, ARTICULO, COD_FABRICANTE, PESO, CATEGORIA, FECHA_PEDIDO),
	CONSTRAINT ped2_fabri_fabri_fk FOREIGN KEY (COD_FABRICANTE) 
	REFERENCES fabricantes2 (Cod_fabricante) ON DELETE CASCADE ON UPDATE CASCADE, 
	CONSTRAINT ped2_ped_c CHECK (UNIDADES_PEDIDAS>0), 
	CONSTRAINT ped2_cat_c CHECK (CATEGORIA IN ("Primera", "Segunda", "Tercera")),
	CONSTRAINT ped2_art_fk FOREIGN KEY (ARTICULO) 
	REFERENCES articulos2 (articulo) ON DELETE CASCADE ON UPDATE CASCADE, 
	CONSTRAINT ped2_codfabri_fk FOREIGN KEY (COD_FABRICANTE) 
	REFERENCES articulos2 (cod_fabricante) ON DELETE CASCADE ON UPDATE CASCADE, 
	CONSTRAINT ped2_peso_fk FOREIGN KEY (PESO) 
	REFERENCES articulos2 (peso) ON DELETE CASCADE ON UPDATE CASCADE, 
	CONSTRAINT ped2_cat_fk FOREIGN KEY (CATEGORIA) 
	REFERENCES articulos2 (categoria_art) ON DELETE CASCADE ON UPDATE CASCADE 
	/*CONSTRAINT ped2_nif_fk FOREIGN KEY (NIF) 
	REFERENCES tiendas1(cod) ON DELETE CASCADE ON UPDATE CASCADE*/ 
	);
